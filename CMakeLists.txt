cmake_minimum_required(VERSION 3.18)
project(PanoramaStitching LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PANORAMA_WITH_CUDA "Build CUDA V1, V2, V3 (requires toolkit)" OFF)
include(CTest)
set(PIPELINE_SOURCES
    src/integration/v0_pipeline.cpp src/integration/v0_selftest.cpp
    src/stage_implementations/v0_stage12.cpp src/stage_implementations/v0_stage34.cpp
    src/stage_implementations/v0_stage5.cpp src/stage_implementations/v0_stage6.cpp
    src/libraries/stb_impl.cpp)
if(PANORAMA_WITH_CUDA)
    enable_language(CUDA)
    list(APPEND PIPELINE_SOURCES src/cuda/cuda_pipeline.cu)
endif()
add_library(panorama_core STATIC ${PIPELINE_SOURCES})
target_include_directories(panorama_core PUBLIC src src/core_headers src/libraries)
if(PANORAMA_WITH_CUDA)
    target_compile_definitions(panorama_core PUBLIC PANORAMA_WITH_CUDA=1)
    set_target_properties(panorama_core PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_options(panorama_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
endif()
function(panorama_tool NAME SOURCE)
    add_executable(${NAME} ${SOURCE})
    target_link_libraries(${NAME} PRIVATE panorama_core)
endfunction()
panorama_tool(stitch src/integration/main.cpp)
panorama_tool(stage12_selftest tools/run_stage12_selftest.cpp)
panorama_tool(stage34_selftest tools/stage34_selftest.cpp)
panorama_tool(stage6_selftest tools/stage6_selftest.cpp)
panorama_tool(synthetic_homography_test tools/synthetic_homography_test.cpp)
panorama_tool(synthetic_pipeline_test tools/synthetic_pipeline_test.cpp)
panorama_tool(debug_keypoints tools/debug_keypoints.cpp)
panorama_tool(timing_stage12 tools/timing_stage12.cpp)
panorama_tool(cuda_host_math_selftest src/cuda/host_math_selftest.cpp)
if(BUILD_TESTING)
    add_test(NAME stage12 COMMAND stage12_selftest)
    add_test(NAME stage34 COMMAND stage34_selftest)
    add_test(NAME stage6 COMMAND stage6_selftest)
    add_test(NAME homography COMMAND synthetic_homography_test)
    add_test(NAME pipeline COMMAND synthetic_pipeline_test ${CMAKE_CURRENT_BINARY_DIR}/out/ctest_v0)
    add_test(NAME cuda_host_math COMMAND cuda_host_math_selftest)
    if(PANORAMA_WITH_CUDA)
        foreach(VERSION RANGE 1 3)
            add_test(NAME cuda_stages_v${VERSION} COMMAND stitch --selftest --version ${VERSION})
            add_test(NAME cuda_pipeline_v${VERSION} COMMAND synthetic_pipeline_test ${CMAKE_CURRENT_BINARY_DIR}/out/ctest_v${VERSION} --version ${VERSION})
        endforeach()
    endif()
endif()
